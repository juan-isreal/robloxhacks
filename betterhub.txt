local Players = game:GetService("Players")
local Lighting = game:GetService("Lighting")
local Workspace = game:GetService("Workspace")
local LocalPlayer = Players.LocalPlayer

-- Allowed prefixes
local PREFIXES = {".", ";"}

local function getPrefixMatch(message)
    for _, prefix in ipairs(PREFIXES) do
        if message:sub(1, #prefix) == prefix then
            return prefix
        end
    end
    return nil
end

print("---------------------------------------------")
print("👑 HD Admin (Sandbox Edition) Loaded!")
print("Supported Prefixes: '.' or ';'")
print("Type .help or ;help in chat for commands.")
print("---------------------------------------------")

-- Listen to chat messages from the local player
LocalPlayer.Chatted:Connect(function(message)
    local usedPrefix = getPrefixMatch(message)
    if not usedPrefix then return end
    
    -- Strip prefix and split message into arguments
    local cmdString = message:sub(#usedPrefix + 1)
    local args = {}
    for arg in cmdString:gmatch("%S+") do
        table.insert(args, arg:lower())
    end
    
    local cmd = args[1]
    if not cmd then return end
    
    local char = LocalPlayer.Character
    local humanoid = char and char:FindFirstChildOfClass("Humanoid")
    
    -- Command Handlers
    if cmd == "speed" or cmd == "ws" then
        local val = tonumber(args[2]) or 16
        if humanoid then
            humanoid.WalkSpeed = val
            print("[Admin] WalkSpeed set to " .. val)
        end
        
    elseif cmd == "jump" or cmd == "jp" then
        local val = tonumber(args[2]) or 50
        if humanoid then
            humanoid.UseJumpPower = true
            humanoid.JumpPower = val
            print("[Admin] JumpPower set to " .. val)
        end
        
    elseif cmd == "gravity" or cmd == "grav" then
        local val = tonumber(args[2]) or 196.2
        Workspace.Gravity = val
        print("[Admin] Gravity set to " .. val)
        
    elseif cmd == "god" then
        if humanoid then
            humanoid.MaxHealth = math.huge
            humanoid.Health = math.huge
            print("[Admin] God Mode Enabled")
        end
        
    elseif cmd == "ungod" then
        if humanoid then
            humanoid.MaxHealth = 100
            humanoid.Health = 100
            print("[Admin] God Mode Disabled")
        end
        
    elseif cmd == "invisible" or cmd == "inv" or cmd == "invis" then
        if char then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") or part:IsA("Decal") then
                    part.Transparency = 1
                end
            end
            print("[Admin] You are now invisible")
        end
        
    elseif cmd == "visible" or cmd == "vis" then
        if char then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.Transparency = 0
                elseif part:IsA("Decal") then
                    part.Transparency = 0
                end
            end
            print("[Admin] You are now visible")
        end
        
    elseif cmd == "time" then
        local val = tonumber(args[2])
        if val then
            Lighting.ClockTime = val
            print("[Admin] Time set to " .. val)
        end
