-- Helper Function to Create Admin Buttons
local function addCommandButton(buttonText, callback)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, 0, 0, 35)
    btn.BackgroundColor3 = Color3.fromRGB(28, 28, 38)
    btn.BorderSizePixel = 0
    btn.TextColor3 = Color3.fromRGB(210, 210, 225)
    btn.Text = "  ⚡ " .. buttonText
    btn.Font = Enum.Font.GothamMedium
    btn.TextSize = 13
    btn.TextXAlignment = Enum.TextXAlignment.Left
    btn.Parent = scroll

    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 6)
    btnCorner.Parent = btn

    btn.MouseButton1Click:Connect(function()
        pcall(callback)
    end)
end

-- ==========================================
-- ADMIN COMMANDS
-- ==========================================

addCommandButton("Super Speed (WalkSpeed = 60)", function()
    local char = LocalPlayer.Character
    if char and char:FindFirstChildOfClass("Humanoid") then
        char:FindFirstChildOfClass("Humanoid").WalkSpeed = 60
    end
end)

addCommandButton("Normal Speed (WalkSpeed = 16)", function()
    local char = LocalPlayer.Character
    if char and char:FindFirstChildOfClass("Humanoid") then
        char:FindFirstChildOfClass("Humanoid").WalkSpeed = 16
    end
end)

addCommandButton("High Jump (JumpPower = 120)", function()
    local char = LocalPlayer.Character
    if char and char:FindFirstChildOfClass("Humanoid") then
        local hum = char:FindFirstChildOfClass("Humanoid")
        hum.UseJumpPower = true
        hum.JumpPower = 120
    end
end)

addCommandButton("Normal Jump (JumpPower = 50)", function()
    local char = LocalPlayer.Character
    if char and char:FindFirstChildOfClass("Humanoid") then
        local hum = char:FindFirstChildOfClass("Humanoid")
        hum.UseJumpPower = true
        hum.JumpPower = 50
    end
end)

addCommandButton("God Mode (Infinite Health)", function()
    local char = LocalPlayer.Character
    if char and char:FindFirstChildOfClass("Humanoid") then
        local hum = char:FindFirstChildOfClass("Humanoid")
        hum.MaxHealth = math.huge
        hum.Health = math.huge
    end
end)

addCommandButton("Ghost Mode (Invisibility)", function()
    local char = LocalPlayer.Character
    if char then
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") or part:IsA("Decal") then
                part.Transparency = 1
            end
        end
    end
end)

addCommandButton("Un-Ghost (Visible)", function()
    local char = LocalPlayer.Character
    if char then
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.Transparency = 0
            elseif part:IsA("Decal") then
                part.Transparency = 0
            end
        end
    end
end)

addCommandButton("Low Gravity (Gravity = 30)", function()
    Workspace.Gravity = 30
end)

addCommandButton("Normal Gravity (Gravity = 196.2)", function()
    Workspace.Gravity = 196.2
end)

addCommandButton("Time: Midnight (Dark Theme)", function()
    Lighting.ClockTime = 0
end)

addCommandButton("Time: Noon (Bright Day)", function()
    Lighting.ClockTime = 14
end)

addCommandButton("Spawn Neon Platform At Feet", function()
    local char = LocalPlayer.Character
    if char and char:FindFirstChild("HumanoidRootPart") then
        local part = Instance.new("Part")
        part.Size = Vector3.new(6, 1, 6)
        part.Position = char.HumanoidRootPart.Position - Vector3.new(0, 3, 0)
        part.Anchored = true
        part.BrickColor = BrickColor.random()
        part.Material = Enum.Material.Neon
        part.Parent = Workspace
    end
end)
