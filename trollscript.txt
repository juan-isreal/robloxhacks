local player = game.Players.LocalPlayer
local playergui = player:WaitForChild("PlayerGui")
local screengui = Instance.new("ScreenGui")
local character = player.Character
local humanoid = character:WaitForChild("Humanoid")
screengui.Parent = playergui
-- gui
local frame = Instance.new("Frame")
frame.Size = UDim2.new(0, 200, 0, 50)
frame.Position = UDim2.new(0.5, -100, 0.1, 0)
frame.Parent = screengui

local textLabel = Instance.new("TextButton")
textLabel.Size = UDim2.new(0, 200, 0, 50)
textLabel.Position = UDim2.new(0.5, -100, 0.1, 0) -- Centered near top
textLabel.Text = "Click me"
textLabel.TextScaled = true
textLabel.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
textLabel.Parent = frame
-- troll hackers
local bad = Instance.new("TextLabel")
bad.Size = UDim2.new(0, 200, 0, 50)
bad.Position = UDim2.new(0.5, -100, 0.1, 0) -- Centered near top
bad.Text = "stop hacking!!1!!1"
bad.TextScaled = true
bad.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
bad.TextColor3 = Color3.fromRGB(255, 255, 255)
bad.Parent = frame
bad.Visible = false

-- Function to run on click
local function onButtonClicked()
  humanoid.Health = 0
  textLabel.Visible = false
  bad.Visible = true
end

textLabel.Activated:Connect(onButtonClicked)
